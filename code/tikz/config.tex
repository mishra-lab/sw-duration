% period diagram
\def\ts{9.5}\def\ti{1}\def\tf{18} % times: survey, fade in, fade out
\newcommand{\period}[4][]{% [style]{start time}{duration}{y}
  \def\to{#2+.01}\def\tx{#2+#3-.01}
  \ifnum1=\fpeval{(\to>\ts)|(\tx<\ts)}
    \draw[perf,ci,uno,#1] (\to,#4) -- (\tx,#4); % inactive
  \else
    \draw[pero,ca,obs,#1] (\to,#4) -- (\ts,#4); % observed active
    \draw[perx,ca,uno,#1] (\ts,#4) -- (\tx,#4); % unobserved active
  \fi}
\newcommand{\periods}[1][]{%
  \begin{scope}
  \clip (\ti,0) rectangle (\tf,6);
  \foreach \y/\dur/\n in {1/1/18,2/2/9,3/3/6,4/6/3,5/12/2}{
    \foreach \i in {0,...,\n}{
      \period[#1]{\i*\dur-\dur+\y}{\dur}{\y} }}
  \draw[cs,per,opacity=.5] (\ts,.5) -- (\ts,5.5);
  \fill[white,path fading=east] (\ti,0.5) rectangle ++(+2,5);
  \fill[white,path fading=west] (\tf,0.5) rectangle ++(-2,5);
  \end{scope}}
