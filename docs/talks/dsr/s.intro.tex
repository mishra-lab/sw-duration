\begin{frame}{Background: duration in sex work}\let\emph\textbf
  \begin{itemize}
    \item Sex work: transient period with \emph{higher risk for STI/HIV} transmission \cite{Scorgie2012}
    \item Mean duration in sex work: key epidemiological quantity and input to STI/HIV models \cite{Garnett2011}
       \\ \eg typically define \emph{rate of exit} from sex work as: \emph{1 / mean duration}
    \item At equilibrium: exiting sex workers replaced by entering sex workers \rarr \emph{"turnover"}
       \\[1ex] {\centering\includegraphics[scale=1]{turnover}\par}
    \item Many other durations similarly important
       \\ \eg transient health states, exposure to risk/protective factors
  \end{itemize}
\end{frame}
