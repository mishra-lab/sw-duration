\section{Introduction}
\begin{itemize}
  \item Sex work is often a transient period with higher risk for STI/HIV transmission \cite{Scorgie2012}
  \item Mean duration in sex work is a key quantity and input to STI/HIV models
  \item Usually, we only observe \emph{right-censored} durations among \emph{active} sex workers \cite{Fazito2012}
  \item However, we want \emph{total} durations among \emph{all} (future/active/past) sex workers
   \par $\rightarrow$ Want to know how exactly these durations relate to each other
  \item Existing methods from a prior review \cite{Fazito2012} have major limitations (Table~\ref{tab:meth})
\end{itemize}

\section{Objectives}
\begin{enumerate}
  \item Derive the exact relationship between
        total durations in sex work overall $f(x)$ and
        right-censored durations observed in cross-sectional studies $f(z|s)$
  \item Develop a Bayesian model of this relationship,
        apply it to aggregate data from a prior global review,\cite{Fazito2012}
        and compare model-estimated mean durations in sex work
        to previously published estimates
\end{enumerate}
