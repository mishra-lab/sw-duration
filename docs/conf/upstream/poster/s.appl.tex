\section{Objective~2: Bayesian Global Synthesis}
\mp{.64}{
\paragraph{Approach}
\begin{itemize}
  \item Coded equations in Stan, with 5 candidate $f(x)$
   \par families, log-uniform parameter priors
  \item Fit model to aggregate duration data in review \cite{Fazito2012}
   \par reflecting 58,249 female sex workers in 5 regions
  \item Resampled posteriors across families per region
\end{itemize}
\paragraph{Model Results}
\begin{itemize}
  \item Inferred >~50\% shorter mean durations vs review
   \par (Table~\ref{tab:appl}) as review methods ignored Steps 1~\&~3
  \item Posterior durations: precise for observed $f(z|s)$
   \par but less precise for target $f(x)$ (Figure~\ref{fig:meta})
  \item Inferred >~50\% women sell sex for <~1 year
   \par in most regions (Figure~\ref{fig:meta})
\end{itemize}}
