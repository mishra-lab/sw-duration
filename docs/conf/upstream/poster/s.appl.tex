\section{Objective~2: Bayesian Global Synthesis}
\mp{.64}{
\paragraph{Approach}
\begin{itemize}
  \item Coded equations in Stan, with 5 candidate $f(x)$
   \par distribution families, uniform parameter priors
  \item Fit model to aggregate sex work duration data
   \par from review\cite{Fazito2012}, keeping best family per region
\end{itemize}
\paragraph{Model Results}
\begin{itemize}
  \item >~2$\times$ shorter mean durations vs review (Table~\ref{tab:appl})
   \par as review ignored length-biased sampling
  \item Posterior durations: precise for observed $f(z|s)$
   \par but less precise for target $f(x)$ (Figure~\ref{fig:meta})
  \item >~50\% women sell sex for <~1 year in most regions
   \par (Figure~\ref{fig:meta})
\end{itemize}}
